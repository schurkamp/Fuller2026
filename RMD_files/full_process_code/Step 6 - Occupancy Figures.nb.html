# Function to read and label data
read_occ_data <- function(path, source_label, pattern) {
  files <- list.files(path, pattern, full.names = TRUE)
  bind_rows(lapply(files, function(file) {
    dat <- read_csv(file)
    species <- str_extract(basename(file), "^[^_]+")  # Extract species from filename
    dat %>%
      mutate(Species = species,
             Source = source_label)
  }))
}

species_occ_comp <- function(species_code, species_name, data) {
  data <- data %>%
    filter(Species == species_code)
  
  data$Route <- as.factor(data$Route)
  data$Year <- as.factor(data$Year)
  
  mocc_comp <- ggplot(data, aes(x = Year, y = mocc, color = Source, group = Source)) +
  geom_errorbar(
    aes(
      ymin = pmax(0, mocc - mean_se),
      ymax = pmin(1, mocc + mean_se)
    ),
    position = position_dodge(width = 0.5),
    width = 0.2
  ) +
  geom_point(size = 3, position = position_dodge(width = 0.5)) +
  
  facet_wrap(~ Route, ncol = 4) +
  scale_color_manual(
    values = c("ARU" = "#F8766D", "PC" = "#00BFC4"),
    name = "Survey Type"
  ) +
  labs(
    title = paste("Mean", species_name, " Occupancy ± SE by Route, Year, and Survey Type"),
    x = "Year",
    y = "Mean Occupancy"
  ) +
  coord_cartesian(ylim = c(0, 1)) +
  theme_grey(base_size = 14) +
  theme(
    strip.text = element_text(face = "bold"),
    legend.position = "top"
  )
  
}
